# showcase (public store) — copy to .env.production before `npm run build`.
# Create React App only reads variables that start with REACT_APP_.
# 2026-10-31 domains: the store must call the APIs on ITS OWN domain (the login cookie
# is shared only inside one domain). Example for store.aleph-k.com
# (for a store on aleph-k.online use auth.aleph-k.online / api.aleph-k.online):
REACT_APP_API_USERS=https://auth.aleph-k.com
REACT_APP_API_SHAO=https://api.aleph-k.com
REACT_APP_SHOWCASE_LINK=https://store.aleph-k.com
REACT_APP_WHATSAPP=+50600000000

# CAPTCHA — Cloudflare Turnstile Site key (public, 2026-10-25). Required in production
# when the backends have TURNSTILE_SECRET_KEY. On localhost / LAN addresses the app uses
# Cloudflare's test key (1x00000000000000000000AA) automatically, so leave it empty there.
REACT_APP_TURNSTILE_SITE_KEY=

# Login buttons (2026-10-26). Values: active | soon | hidden
#   active = the button works normally
#   soon   = shown with "Coming soon"; clicking only shows a message (no login code runs)
#   hidden = not shown
# Missing line = active. A misspelled value hides the button.
# Keep Facebook/Apple DISABLED in Firebase Authentication while they are "soon".
REACT_APP_AUTH_GOOGLE=active
REACT_APP_AUTH_EMAIL=active
REACT_APP_AUTH_MICROSOFT=active
REACT_APP_AUTH_FACEBOOK=soon
REACT_APP_AUTH_APPLE=soon
# Optional: custom message for "soon" buttons (default: Spanish or English by app language)
